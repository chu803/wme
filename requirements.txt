# ------------------------------------------------------------------------------
# 【依赖清单说明】本项目提供三份清单，按需求选用：
#   requirements.txt     —— 完整版（默认）：含 CPU 版 torch + LaMa + 全部能力。
#   requirements-min.txt —— 最小版：不装 torch/LaMa，仅 OpenCV 降级引擎，
#                          安装体积最小，适合无显卡、只想快速体验的用户。
#   requirements-gpu.txt —— 增强版：在完整版基础上把 torch 换成 CUDA 版，
#                          用于启用 NVIDIA 显卡加速（详见该文件头部说明）。
# 普通用户直接装 requirements.txt 即可；想更小体积选 -min；有显卡想提速选 -gpu。
# ------------------------------------------------------------------------------


# ------------------------------------------------------------------------------
# 【第一组】深度学习推理框架 —— LaMa 模型运行的地基
# ------------------------------------------------------------------------------
# torch：PyTorch 主框架，负责加载 big-lama.pt 权重并执行神经网络推理。
#        锁定 2.3.1 是因为该版本对 TorchScript 的兼容性最稳，且支持 Python 3.9~3.12。
torch==2.3.1

# torchvision：提供图像张量变换工具（归一化、插值、格式转换）。
#              版本号必须与 torch 严格配套，否则会出现 C++ 符号找不到的报错。
torchvision==0.18.1


# ------------------------------------------------------------------------------
# 【第二组】图像处理核心库 —— 掩码生成、备选修复算法、格式读写
# ------------------------------------------------------------------------------
# opencv-python：OpenCV 主库。用途有三：
#                1. 备选轻量修复方案（cv2.inpaint 的 TELEA / NS 算法）
#                2. 智能水印自动检测（边缘检测、形态学、MSER 文字区域探测）
#                3. 掩码后处理（膨胀、腐蚀、高斯羽化）
opencv-python==4.10.0.84

# opencv-contrib-python：OpenCV 扩展模块，提供 xphoto 高级修复算法与
#                        更完善的文本检测器，作为自动检测模块的增强能力。
opencv-contrib-python==4.10.0.84

# numpy：所有图像在内存中的通用载体（H×W×C 数组），全项目基础依赖。
#        锁定 1.26.x 是因为 numpy 2.x 与部分 opencv/torch 组合仍存在 ABI 冲突。
numpy==1.26.4

# Pillow：图片文件读写与格式转换（PNG/JPG/WebP），负责 EXIF 方向校正、
#         PNG 透明通道保留、输出画质（quality）控制。
Pillow==10.4.0

# scikit-image：提供高质量的形态学重建、局部方差计算、结构相似度（SSIM）评估，
#               用于半透明水印检测与修复质量自评。
scikit-image==0.24.0

# scipy：科学计算基础库，提供距离变换、卷积滤波、连通域分析等底层算子。
scipy==1.14.1


# ------------------------------------------------------------------------------
# 【第三组】Web 可视化界面框架
# ------------------------------------------------------------------------------
# gradio：网页 UI 框架，提供画笔涂抹组件（ImageEditor）、图片对比滑块、
#         文件批量上传、实时进度条等全部交互能力。
#         统一使用 6.x：本项目的界面代码（ui/app.py、ui/tab_single.py、
#         core/mask_generator.py）已按 Gradio 6.x 实测签名编写——
#         6.x 中 ImageEditor 返回值改为 dict(background/layers/composite)、
#         theme/css 移入 launch()、show_api 改为 footer_links 控制。
#         若改用 4.x，会因 launch(theme=...) 等签名差异而启动失败。
#         阶段 188：补上**上限** `<7`——上述说明明确要求「统一使用 6.x」，
#         但原先只写 `>=6.0` 会允许 7.x 装上（Gradio 大版本破坏性变更频繁），
#         即「钉版与自述意图不一致」。上限使注释与约束一致。
gradio>=6.0,<7

# gradio_client：Gradio 的配套客户端库，版本由 gradio 主体自动匹配，
#               无需单独锁定（避免与 gradio 6.x 要求的版本冲突）。


# ------------------------------------------------------------------------------
# 【第四组】模型下载与网络请求
# ------------------------------------------------------------------------------
# requests：download_model.py 用它以流式方式下载 LaMa 权重，支持断点续传。
requests==2.32.3

# tqdm：命令行下载进度条，让用户清楚看到模型下载了百分之多少。
tqdm==4.66.5

# urllib3：requests 的底层依赖，显式锁定版本避免 SSL 警告刷屏。
urllib3==2.2.3

# certifi：HTTPS 根证书集合，解决 Windows 上常见的 SSL 证书验证失败问题。
certifi==2024.8.30


# ------------------------------------------------------------------------------
# 【第五组】系统资源监控与硬件自适应
# ------------------------------------------------------------------------------
# psutil：读取物理内存总量与可用量，用于大图内存溢出防护、
#         自动决定是否启用分块（Tile）推理。
psutil==6.0.0

# py-cpuinfo：读取 CPU 型号与核心数，在 CPU 模式下自动设置最优线程数。
py-cpuinfo==9.0.0


# ------------------------------------------------------------------------------
# 【第六组】日志、配置与工程化工具
# ------------------------------------------------------------------------------
# colorama：让 Windows 命令行窗口正确显示彩色日志（红色报错、黄色警告）。
colorama==0.4.6

# python-dotenv：支持通过项目根目录的 .env 文件覆盖默认配置，
#                方便用户不改代码就能修改端口、模型路径等参数。
python-dotenv==1.0.1

# pydantic：配置项的类型校验，防止用户在配置文件里填入非法数值导致程序崩溃。
pydantic==2.9.2


# ------------------------------------------------------------------------------
# 【第七组】打包分发（仅在生成 EXE 时需要，日常运行可不装）
# ------------------------------------------------------------------------------
# pyinstaller：将整个项目打包为 Windows 独立可执行程序，配合 build.spec 使用。
pyinstaller==6.10.0

# pyinstaller-hooks-contrib：社区维护的第三方库打包钩子集合，
#                            解决 torch / gradio 打包后缺文件的经典问题。
pyinstaller-hooks-contrib==2024.8
